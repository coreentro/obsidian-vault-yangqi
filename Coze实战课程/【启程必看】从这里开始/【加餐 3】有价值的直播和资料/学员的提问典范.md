## 代码块

```Markdown
from typing import Dict, Any
import dashscope
from dashscope.audio.asr import Transcription
import json
import requests

def handler(args) -> Dict[str, Any]:
    """
    调用阿里云Paraformer语音识别API的Coze代码节点
    
    Args:
        args: Coze平台的参数对象
            args.input.api_key: 阿里云API Key
            args.input.file_urls: 音频文件URL(字符串或列表)
            args.input.model: 可选，模型名称，默认 paraformer-v2
            args.input.language_hints: 可选，语言提示，默认 zh
    
    Returns:
        Dict[str, Any]: 包含识别结果的字典
            success: 是否成功
            message: 处理消息
            task_id: 任务ID
            task_status: 任务状态
            results: 识别结果列表
    """
    try:
        # 获取参数
        api_key = args.input.api_key
        file_url = args.input.file_urls
        model = getattr(args.input, 'model', 'paraformer-v2')
        language_hint = getattr(args.input, 'language_hints', 'zh')
        
        # 设置API Key
        dashscope.api_key = api_key
        
        # 确保 file_urls 是列表格式
        if isinstance(file_url, str):
            file_urls = [file_url]
        elif isinstance(file_url, list):
            file_urls = file_url
        else:
            return {
                'success': False,
                'message': 'Invalid file_urls format',
                'task_id': '',
                'task_status': 'FAILED',
                'results': []
            }
            
        # 确保 language_hints 是列表格式
        if isinstance(language_hint, str):
            language_hints = [language_hint]
        elif isinstance(language_hint, list):
            language_hints = language_hint
        else:
            language_hints = ['zh']
        
        # 提交任务
        response = Transcription.async_call(
            model=model,
            file_urls=file_urls,
            language_hints=language_hints
        )
        
        # 检查任务提交是否成功
        if response.status_code == 200 and hasattr(response.output, 'task_id'):
            task_id = response.output.task_id
            print(f"Task ID: {task_id}")
            
            # 等待结果
            wait_response = Transcription.wait(task=task_id)
            
            # 打印完整的响应内容以便调试
            print("=== Complete Response ===")
            print(f"Status Code: {wait_response.status_code}")
            print(f"Task Status: {wait_response.output.task_status}")
            print("Response Output:")
            print(json.dumps(wait_response.output.__dict__, indent=2))
            
            # 检查任务状态
            if (wait_response.status_code == 200 and 
                wait_response.output.task_status == 'SUCCEEDED' and
                hasattr(wait_response.output, 'results')):
                
                results = []
                # 处理每个音频文件的结果
                for result in wait_response.output.results:
                    if isinstance(result, dict) and result.get('transcription_url'):
                        try:
                            # 获取识别结果
                            transcript_response = requests.get(result['transcription_url'])
                            if transcript_response.status_code == 200:
                                transcript_data = transcript_response.json()
                                # 从 transcripts 字段获取文本
                                if 'transcripts' in transcript_data:
                                    text = transcript_data['transcripts'][0].get('text', '')
                                    if text:
                                        results.append({
                                            'file_url': result.get('file_url', file_urls[0]),
                                            'text': text,
                                            'status': 'SUCCEEDED'
                                        })
                        except Exception as e:
                            print(f"Error getting transcript: {str(e)}")
                            continue
                
                return {
                    'success': bool(results),
                    'message': 'Recognition completed successfully' if results else 'No text content found',
                    'task_id': task_id,
                    'task_status': wait_response.output.task_status,
                    'results': results
                }
        
        return {
            'success': False,
            'message': 'Failed to submit task',
            'task_id': '',
            'task_status': 'FAILED',
            'results': []
        }
            
    except Exception as e:
        print(f"Error occurred: {str(e)}")
        return {
            'success': False,
            'message': f"Error occurred: {str(e)}",
            'task_id': '',
            'task_status': 'FAILED',
            'results': []
        } 
```
